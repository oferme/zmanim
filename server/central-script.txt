const API_URL = 'https://script.google.com/macros/s/AKfycbzk_q8nwWNNXzr2UYee4WPWa0k6D-ZGegG16g5KFU46jew3l_bKigET6eW7-cLwTVklyg/exec';

const SHEET = 'בתי כנסת';
// כל שורה = "היכל" אחד: הגדרות, הודעות וקישור משלו. כמה היכלות של אותו בית כנסת – אותו שם בעמודה "ארגון".
// עמודות: מזהה | שם | קישור למסמך | פעיל | גרסה | מייל גבאי | קישור למסך | הערות | מפתח | הגדרות | קישור להגדרות | ארגון | מספר מסכים
const HEAD = ['מזהה', 'שם (היכל)', 'קישור למסמך', 'פעיל', 'גרסה', 'מייל גבאי', 'קישור למסך', 'הערות', 'מפתח', 'הגדרות', 'קישור להגדרות', 'ארגון', 'מספר מסכים', 'נוסח'];
// מיקום כל עמודה נקבע לפי הכותרת שלה בשורה 1 (כך שאפשר להזיז / להוסיף עמודות בגיליון בלי לשבור את המערכת)
const COL_KEYS = ['id', 'name', 'doc', 'active', 'ver', 'email', 'display', 'notes', 'key', 'settings', 'settingsLink', 'org', 'screens', 'nusach'];
let C = { id: 0, name: 1, doc: 2, active: 3, ver: 4, email: 5, display: 6, notes: 7, key: 8, settings: 9, settingsLink: 10, org: 11, screens: 12, nusach: 13 };
function mapCols_(sh) {
  const last = Math.max(sh.getLastColumn(), 1);
  const h = sh.getRange(1, 1, 1, last).getValues()[0].map(x => String(x).replace(/\s+/g, ' ').trim());
  const m = {}; let next = last;
  COL_KEYS.forEach((k, i) => {
    let j = h.indexOf(HEAD[i]);
    if (j < 0 && k === 'name') j = h.findIndex(x => /^שם/.test(x));
    if (j < 0) { j = next++; sh.getRange(1, j + 1).setValue(HEAD[i]).setFontWeight('bold'); h[j] = HEAD[i]; }   // עמודה חסרה – נוספת בסוף
    m[k] = j;
  });
  C = m;
  return sh;
}
// בונה שורה חדשה לפי מיקומי העמודות
function rowOf_(vals) {
  const arr = new Array(Math.max.apply(null, COL_KEYS.map(k => C[k])) + 1).fill('');
  Object.keys(vals).forEach(k => { arr[C[k]] = vals[k]; });
  return arr;
}

// ---------- גיבוי ----------
// עותק של הגיליון (כולל הקוד שמחובר אליו) לתיקייה "גיבויים – לוח זמנים" בדרייב. נשמרים 30 העותקים האחרונים.
// בכל יום ראשון נשלח גם קובץ Excel של הגיליון למייל של בעל הסקריפט – עותק מחוץ לדרייב.
const BACKUP_FOLDER = 'גיבויים – לוח זמנים';
function backupFolder_() {
  const it = DriveApp.getFoldersByName(BACKUP_FOLDER);
  return it.hasNext() ? it.next() : DriveApp.createFolder(BACKUP_FOLDER);
}
function makeBackup_(sendMail) {
  const ss = sheet_().getParent(), folder = backupFolder_();
  const stamp = Utilities.formatDate(new Date(), 'Asia/Jerusalem', 'yyyy-MM-dd HH-mm');
  DriveApp.getFileById(ss.getId()).makeCopy('גיבוי ' + stamp + ' – ' + ss.getName(), folder);
  const files = [], it = folder.getFiles();
  while (it.hasNext()) files.push(it.next());
  files.sort((a, b) => b.getDateCreated() - a.getDateCreated());
  files.slice(30).forEach(f => { try { f.setTrashed(true); } catch (e) {} });
  if (sendMail) {
    const url = 'https://docs.google.com/spreadsheets/d/' + ss.getId() + '/export?format=xlsx';
    const blob = UrlFetchApp.fetch(url, { headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() } }).getBlob()
      .setName('גיבוי לוח זמנים ' + stamp + '.xlsx');
    MailApp.sendEmail(Session.getEffectiveUser().getEmail(), 'גיבוי שבועי – ניהול בתי כנסת', 'מצורף גיבוי של גיליון הניהול (' + stamp + ').\nכדאי לשמור את המייל.', { attachments: [blob] });
  }
  return stamp;
}
function backupNow() {
  const stamp = makeBackup_(true);
  SpreadsheetApp.getUi().alert('הגיבוי נשמר (' + stamp + ') בתיקייה "' + BACKUP_FOLDER + '" בדרייב, ונשלח עותק Excel למייל שלך.');
}
function backupNightly() { makeBackup_(new Date().getDay() === 0); }   // כל לילה; ביום ראשון – גם למייל
function enableBackup() {
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'backupNightly').forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('backupNightly').timeBased().everyDays(1).atHour(2).inTimezone('Asia/Jerusalem').create();
  SpreadsheetApp.getUi().alert('הגיבוי האוטומטי הופעל: כל לילה בסביבות 2:00 נשמר עותק בדרייב, ובכל יום ראשון נשלח גם קובץ Excel למייל שלך.');
}

function setup() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  PropertiesService.getScriptProperties().setProperty('SS_ID', ss.getId());
  let sh = ss.getSheetByName(SHEET);
  if (!sh) {
    sh = ss.insertSheet(SHEET);
    sh.appendRow(HEAD);
    sh.setFrozenRows(1);
    sh.setRightToLeft(true);
  }
  mapCols_(sh);   // לא דורס כותרות קיימות – רק מוסיף עמודות חסרות
  let dv = ss.getSheetByName(DEV_SHEET);
  if (!dv) { dv = ss.insertSheet(DEV_SHEET); dv.appendRow(DEV_HEAD); dv.setFrozenRows(1); dv.setRightToLeft(true); dv.getRange(1, 1, 1, DEV_HEAD.length).setFontWeight('bold'); }
  const rule = SpreadsheetApp.newDataValidation().requireValueInList(['מאושר', 'ממתין', 'חסום'], true).build();
  dv.getRange(2, D.status + 1, 1000, 1).setDataValidation(rule);
  SpreadsheetApp.getUi().alert('ההתקנה הסתיימה. עכשיו אפשר להוסיף בתי כנסת מהתפריט "לוח זמנים".');
}

function sheet_() {
  let ss = null;
  try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (e) {}
  if (!ss) ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty('SS_ID'));
  return mapCols_(ss.getSheetByName(SHEET));
}
function devSheet_() { return sheet_().getParent().getSheetByName(DEV_SHEET); }
function findRow_(id) {
  const sh = sheet_(), rows = sh.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) if (String(rows[i][C.id]).trim().toLowerCase() === id) return { sh, r: i + 1, row: rows[i] };
  return null;
}
const newKey_ = () => Utilities.getUuid().replace(/-/g, '').slice(0, 14);
// קישור לאפליקציית ההגדרות (באתר המסך). הטופס הישן ממשיך לעבוד גם כן: API_URL + '?page=settings&shul=...&key=...'
const settingsLink_ = (id, key) => DISPLAY_URL + 'settings.html?shul=' + id + '&key=' + key;
// מספר הגרסה של הקוד הזה. המסך בודק אותו ומתריע אם השרת לא עודכן (לא נפרסה "גרסה חדשה")
const SERVER_BUILD = 19;
function defaults_(name) {
  return { name: name || '', city: 'ירושלים', lat: '', lon: '', tz: 'Asia/Jerusalem', israel: true, candle: 40, havdalah: '',
    useElev: true, elev: '', slotR: ['learn', 'tefila'], rotR: 20, learn: ['daf'], slotL: ['msgs'], slotB: ['zmanim', 'prayers'], rotB: 30,
    rotate: 30, clock: 100, bottom: 90, tachanun: true, remind: 30, prayers: [], theme: 'dark', theme2: 'picture' };
}

// ---------- הוספת בית כנסת ----------
function addShul() {
  const ui = SpreadsheetApp.getUi();
  const ask = (q) => { const r = ui.prompt(q, ui.ButtonSet.OK_CANCEL); return r.getSelectedButton() === ui.Button.OK ? r.getResponseText().trim() : null; };

  const id = (ask('מזהה קצר באנגלית (אותיות, ספרות ומקף), למשל: ohel-moshe') || '').toLowerCase();
  if (!id) return;
  if (!/^[a-z0-9-]+$/.test(id)) { ui.alert('המזהה יכול להכיל רק אותיות באנגלית, ספרות ומקף.'); return; }
  if (findRow_(id)) { ui.alert('המזהה הזה כבר קיים.'); return; }
  const name = ask('שם בית הכנסת');
  if (name === null) return;
  const org = ask('שם בית הכנסת / המוסד (לקיבוץ כמה היכלות יחד). אם זה היכל יחיד – אפשר להשאיר ריק');
  if (org === null) return;
  const scr = ask('כמה מסכים מאושרים להיכל הזה? (מספר)');
  if (scr === null) return;
  const email = ask('מייל (ג׳ימייל) של הגבאי/הרב – אפשר להשאיר ריק');
  if (email === null) return;
  const nusach = ask('נוסח בית הכנסת: אשכנז / ספרד / עדות המזרח – אפשר להשאיר ריק');
  if (nusach === null) return;

  const docUrl = createDoc_(name, email);
  const doc = { getUrl: () => docUrl };

  const key = newKey_(), display = DISPLAY_URL + '?shul=' + id, sLink = settingsLink_(id, key);
  const sh = sheet_();
  sh.appendRow(rowOf_({ id, name, doc: doc.getUrl(), active: 'כן', ver: 1, email, display, key, settings: JSON.stringify(defaults_(name)),
    settingsLink: sLink, org: org || name, screens: Math.max(1, parseInt(scr, 10) || 1), nusach }));

  if (email) {
    try {
      MailApp.sendEmail(email, 'לוח הזמנים של ' + name,
        'שלום,\n\nלעריכת ההגדרות של המסך (מהטלפון או מהמחשב):\n' + sLink +
        '\n\nלכתיבת הודעות למסך:\n' + doc.getUrl() + '\n\nכדאי לשמור את שני הקישורים במסך הבית של הטלפון.');
    } catch (e) {}
  }
  ui.alert('בית הכנסת נוסף!\n\nקישור למסך:\n' + display + '\n\nקישור להגדרות (לגבאי):\n' + sLink + '\n\nמסמך ההודעות:\n' + doc.getUrl());
}

// יוצר מפתח לשורות בלי מפתח, ומעדכן את כל קישורי ההגדרות לכתובת העדכנית (המפתחות הקיימים נשמרים)
function fillMissingLinks() {
  const ui = SpreadsheetApp.getUi();
  const sh = sheet_(), rows = sh.getDataRange().getValues();
  let n = 0, d = 0;
  let fixed = 0, mig = 0;
  for (let i = 1; i < rows.length; i++) {
    // שורה שנוספה בגרסה קודמת לפי סדר עמודות קבוע (המפתח בעמודה I) – מועברת לעמודות הנכונות לפי הכותרות
    const old = rows[i];
    if (C.key !== 8 && !/^[0-9a-f]{14}$/.test(String(old[C.key] || '').trim()) && /^[0-9a-f]{14}$/.test(String(old[8] || '').trim())) {
      const vals = {}; COL_KEYS.forEach((k, j) => { vals[k] = old[j] === undefined ? '' : old[j]; });
      const arr = rowOf_(vals), width = Math.max(arr.length, old.length);
      while (arr.length < width) arr.push('');
      sh.getRange(i + 1, 1, 1, width).setValues([arr]);
      rows[i] = arr; fixed++;
    }
    const id = String(rows[i][C.id]).trim().toLowerCase();
    if (!id) continue;
    let key = String(rows[i][C.key] || '');
    if (!key) { key = newKey_(); sh.getRange(i + 1, C.key + 1).setValue(key); n++; }
    sh.getRange(i + 1, C.settingsLink + 1).setValue(settingsLink_(id, key));
    sh.getRange(i + 1, C.display + 1).setValue(DISPLAY_URL + '?shul=' + id);
    const st = parseJson_(rows[i][C.settings]);
    if (st && migrateDaf_(st)) { sh.getRange(i + 1, C.settings + 1).setValue(JSON.stringify(st)); mig++; }
    if (!String(rows[i][C.doc]).trim()) {   // אין מסמך הודעות – מציע ליצור
      const name = String(rows[i][C.name] || id);
      const ans = ui.alert('ל"' + name + '" אין מסמך הודעות.\n\nליצור עכשיו מסמך הודעות חדש?\n(אם כבר יש לכם מסמך – לחצו "לא" והדביקו את הקישור שלו בעמודה "קישור למסמך")', ui.ButtonSet.YES_NO);
      if (ans === ui.Button.YES) {
        const url = createDoc_(name, String(rows[i][C.email] || '').trim());
        sh.getRange(i + 1, C.doc + 1).setValue(url); d++;
      }
    }
  }
  ui.alert('קישורי ההגדרות עודכנו.' + (fixed ? ' תוקנו ' + fixed + ' שורות שהיו בעמודות לא נכונות.' : '') + (mig ? ' ב-' + mig + ' בתי כנסת "דף יומי" הועבר ל"לימוד יומי".' : '') + (n ? ' נוצרו ' + n + ' מפתחות חדשים.' : '') + (d ? ' נוצרו ' + d + ' מסמכי הודעות.' : ''));
}
function createDoc_(name, email) {
  const doc = DocumentApp.create('הודעות – ' + name);
  const body = doc.getBody();
  TEMPLATE().split('\n').forEach((line, i) => {
    let p;
    if (i === 0) { p = body.getParagraphs()[0]; p.setText(line); }
    else p = body.appendParagraph(line);
    p.setLeftToRight(false);
    p.setAlignment(DocumentApp.HorizontalAlignment.RIGHT);
  });
  doc.saveAndClose();
  if (email) { try { DriveApp.getFileById(doc.getId()).addEditor(email); } catch (e) {} }
  return doc.getUrl();
}

// ---------- השרת ----------
function doGet(e) {
  const p = (e && e.parameter) || {};
  const id = String(p.shul || '').trim().toLowerCase();
  if (p.page === 'settings') return settingsPage_(id, String(p.key || ''));
  if (p.page === 'api') return json_(apiLoad_(id, String(p.key || '')));

  const out = (t) => ContentService.createTextOutput(t).setMimeType(ContentService.MimeType.TEXT);
  if (p.what === 'build') return out('גרסת שרת: ' + SERVER_BUILD);   // בדיקה מהירה בדפדפן: ...exec?what=build
  if (p.what === 'yt') return json_(ytLatest_(String(p.url || ''), String(p.filter || ''), true));   // בדיקת "השיעור האחרון": ...exec?what=yt&url=...&filter=...
  if (!id) return out('ERR:NO_ID');
  const what = String(p.what || '');
  const scr = String(p.screen || '') === '2' ? 2 : 1;   // מסך שני של אותו היכל: ...&screen=2
  const cache = CacheService.getScriptCache();
  if (what === 'hal' || what === 'haltest') {   // פס ההלכות של היום (מספריא, דרך השרת – עובר גם בסינון)
    if (what === 'haltest') return json_(halTest_(String(p.book || 'Mishnah Berurah')));
    const f = findRow_(id); const st = f ? parseJson_(f.row[C.settings]) || {} : {};
    const day = String(p.date || '').match(/^\d{4}-\d{2}-\d{2}$/) ? String(p.date) : Utilities.formatDate(new Date(), 'Asia/Jerusalem', 'yyyy-MM-dd');
    return json_(halToday_(id, cleanHal_(st.hal), day));
  }
  if (what === 'img') {   // תמונה למרכז מסך התמונה – רק תמונה ששייכת להגדרות של בית הכנסת הזה
    const f = findRow_(id); const st = f ? parseJson_(f.row[C.settings]) || {} : {};
    const fid = String(p.id || '');
    if (!fid || !picImages_(st).includes(fid)) return out('');
    try {
      const b = DriveApp.getFileById(fid).getBlob();
      return out('data:' + b.getContentType() + ';base64,' + Utilities.base64Encode(b.getBytes()));
    } catch (e) { return out(''); }
  }
  if (what === 'logo') {   // הלוגו כ-data URL דרך השרת – כך הוא עובר גם במחשבים עם סינון שחוסם את googleusercontent
    const f = findRow_(id); const st = f ? parseJson_(f.row[C.settings]) || {} : {};
    if (!st.logoId) return out('');
    const hitL = cache.get('logo_' + st.logoId); if (hitL !== null) return out(hitL);
    try {
      const b = DriveApp.getFileById(st.logoId).getBlob();
      const d = 'data:' + b.getContentType() + ';base64,' + Utilities.base64Encode(b.getBytes());
      if (d.length < 95000) cache.put('logo_' + st.logoId, d, 21600);
      return out(d);
    } catch (e) { return out(''); }
  }
  if (what !== 'version' && what !== 'zoom') {   // אישור מסך (לא נדרש לבדיקת גרסה ולתוכנת הזום במחשב)
    const dev = String(p.dev || '').replace(/[^a-z0-9]/gi, '').slice(0, 32);
    const err = checkDevice_(id, dev, String(p.code || '').replace(/\D/g, '').slice(0, 6), cache);
    if (err) return out(err);
  }
  const ck = what + 'shul_' + id + (scr === 2 ? '_2' : '');
  const hit = cache.get(ck);
  if (hit !== null) return out(hit);

  const f = findRow_(id);
  if (!f) return out('ERR:NOT_FOUND');
  const row = f.row;
  const ver = String(row[C.ver] || 1).replace(/\D/g, '') || '1';
  if (what === 'version') { cache.put('versionshul_' + id, ver, 60); return out(ver); }
  if (what === 'zoom') {   // לוח השיעורים בזום – נקרא ע"י zoom-helper.ps1 במחשב של בית הכנסת
    // שיעורים בזום / באתר – רק במסך הראשי (מחשב של מסך 2 לא מצטרף)
    const z = String(row[C.active]).trim() === 'לא' || scr === 2 ? '' : zoomLines_(parseJson_(row[C.settings]) || {}, String(row[C.name] || id));
    cache.put(ck, z, 60); return out(z);
  }

  const active = String(row[C.active]).trim();
  let text;
  if (active === 'לא' || active.toLowerCase() === 'false') {
    text = 'ERR:INACTIVE';
  } else {
    text = '';
    const m = String(row[C.doc]).match(/[-\w]{25,}/);
    if (m) {
      const file = DriveApp.getFileById(m[0]);
      text = file.getMimeType() === MimeType.GOOGLE_DOCS
        ? DocumentApp.openById(m[0]).getBody().getText()
        : file.getBlob().getDataAsString('UTF-8');
    }
    const s = parseJson_(row[C.settings]);
    if (s) nusachDefault_(s, row);
    if (s) {   // הגדרות מהטופס גוברות על שורות ! במסמך (חוץ מטבלת נץ)
      text = text.split(/\r?\n/).filter(l => !/^\s*!/.test(l) || /טבלת נץ/.test(l)).join('\n') + '\n' + settingsLines_(s, scr);
    }
    text += '\n! גרסת מערכת: ' + ver + '\n! גרסת שרת: ' + SERVER_BUILD;
  }
  cache.put(ck, text, 60);
  return out(text);
}

function checkDevice_(id, dev, code, cache) {
  if (!dev) return 'ERR:DEVICE:0000';
  const ck = 'dev_' + id + '_' + dev;
  if (cache.get(ck) === '1') return null;
  const lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    const f = findRow_(id);
    if (!f) return 'ERR:NOT_FOUND';
    const dv = devSheet_(), rows = dv.getDataRange().getValues(), now = new Date();
    let approved = 0;
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][D.shul]).toLowerCase() !== id) continue;
      if (String(rows[i][D.dev]) === dev) {
        dv.getRange(i + 1, D.last + 1).setValue(now);
        const st = String(rows[i][D.status]).trim();
        if (st === 'מאושר') { cache.put(ck, '1', 600); return null; }

    }
    const quota = Math.max(1, parseInt(f.row[C.screens], 10) || 1);
    const ok = approved < quota;

    if (ok) { cache.put(ck, '1', 600); return null; }
    return 'ERR:DEVICE:' + (code || '----');
  } finally { lock.releaseLock(); }
}

function resolveZoom_(link) {
  if (/zoom\.(us|com)\/(j|w|wc\/join)\/\d+/i.test(link)) return link;
  const cache = CacheService.getScriptCache(), ck = 'zr_' + Utilities.base64EncodeWebSafe(link).slice(0, 200);
  const hit = cache.get(ck); if (hit !== null) return hit || link;
  let url = link, out = '';
  try {
    for (let i = 0; i < 6 && !out; i++) {
      const r = UrlFetchApp.fetch(url, { followRedirects: false, muteHttpExceptions: true });
      const h = r.getAllHeaders(), loc = h.Location || h.location;
      const found = s => (String(s || '').match(/https:\/\/[\w.-]*zoom\.(?:us|com)\/(?:j|w)\/\d+[^\s"'<>]*/i) || [])[0];
      if (loc) { if (found(loc)) out = found(loc); else url = /^https?:/i.test(loc) ? loc : url.replace(/^(https?:\/\/[^/]+).*$/, '$1') + loc; }
      else { out = found(r.getContentText()) || ''; break; }
    }
  } catch (e) {}
  out = out.replace(/&amp;/g, '&');
  cache.put(ck, out, out ? 21600 : 300);
  return out || link;
}

  if (scr === 1) liveSessions_(s).forEach(z => {   // שיעורים – רק במסך הראשי
    const link = (z.type === 'youtube' && z.latest) ? (ytLatest_(z.link, z.filter) || z.link) : z.link;   // "השיעור האחרון" – הסרטון האחרון שעלה
    L.push('! שיעור חי: ' + z.ds.map(d => ZOOM_DAYS[d]).join('') + ' ' + z.t + (z.e ? '-' + z.e : '') + ' | ' + z.type + ' | ' + z.title + ' | ' + link);
  });
  // תפילה: כל זמן יכול לקבל מקום (היכל) – נכתב בסוגריים אחרי הזמן: 7:15 (בית המדרש)
  (s.prayers || []).filter(p => onScr_(p, scr)).forEach(p => {
    if (!p.label || !p.value) return;
    const halls = Array.isArray(p.halls) ? p.halls : [];
    const value = String(p.value).split(',').map((x, i) => x.trim() + (halls[i] ? ' (' + String(halls[i]).replace(/[(),|]/g, ' ').trim() + ')' : '')).join(', ');
    L.push('! תפילה ' + (p.q ? '[' + p.q + '] ' : '') + String(p.label).replace(/[:\[\]]/g, '') + ': ' + value);
  });
  return L.join('\n');
}

// רישיון שמותר לשימוש מסחרי: נחלת הכלל / CC0 / CC-BY / CC-BY-SA (לא NC / ND / זכויות שמורות)
const okLic_ = l => { l = String(l || ''); return /public\s*domain|cc0|cc[- ]by/i.test(l) && !/nc|nd|copyright/i.test(l.replace(/public\s*domain/i, '')); };
// המהדורה העברית שבה משתמשים לספר (נשמרת במטמון ליום)
function halVersion_(index) {

const stripHtml_ = t => String(t || '').replace(/<sup[^>]*>[\s\S]*?<\/sup>/gi, '').replace(/<i class="footnote"[^>]*>[\s\S]*?<\/i>/gi, '')

const flat_ = a => Array.isArray(a) ? a.reduce((x, y) => x.concat(flat_(y)), []) : (a ? [a] : []);

const heb_ = n => { const o = ['', 'א', 'ב', 'ג', 'ד', 'ה', 'ו', 'ז', 'ח', 'ט'], t = ['', 'י', 'כ', 'ל', 'מ', 'נ', 'ס', 'ע', 'פ', 'צ'], h = ['', 'ק', 'ר', 'ש', 'ת'];
  let s = (n >= 400 ? 'ת' : '') + h[Math.floor((n % 400) / 100)]; n %= 100; s += n === 15 ? 'טו' : n === 16 ? 'טז' : t[Math.floor(n / 10)] + o[n % 10];
  return s.length > 1 ? s.slice(0, -1) + '״' + s.slice(-1) : s + '׳'; };
const daysBetween_ = (a, b) => Math.round((Date.parse(b + 'T12:00:00Z') - Date.parse(a + 'T12:00:00Z')) / 864e5);
// ההלכות של היום: מתחילים בנקודת ההתחלה ביום ההתחלה, ומתקדמים "per" סעיפים בכל יום (ממשיכים לסימן הבא, ובסוף הספר – מההתחלה)
function halToday_(id, h, day) {
  if (!h.on) return { ok: false, error: 'פס ההלכות כבוי' };
  if (h.src === 'text') return { ok: true, title: 'הלכה יומית', items: h.text.split(/\n{2,}/).map(t => ({ label: '', text: t.replace(/\n/g, ' ').trim() })).filter(x => x.text), source: '' };
  const cache = CacheService.getScriptCache();
  const sig = JSON.stringify([h.src, h.key, h.book, h.per, h.start, h.siman, h.seif, h.link, h.date]);
  const ck = 'hald_' + id + '_' + day + '_' + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, sig)).slice(0, 12);
  const hit = cache.get(ck); if (hit) return JSON.parse(hit);
  let out;
  try { out = h.src === 'hebcal' ? halHebcal_(h, day) : halBook_(id, h, day, sig); }
  catch (e) { out = { ok: false, error: String(e && e.message || e) }; }
  const js = JSON.stringify(out);
  if (out.ok && js.length < 95000) cache.put(ck, js, 21600);
  return out;
}
function halBook_(id, h, day, sig) {
  const start = h.date || day;
  const n = daysBetween_(start, day);
  if (n < 0) return { ok: false, error: 'הלימוד מתחיל ב-' + start };
  // נקודת ההתחלה
  let ref = h.book;
  if (h.start === 'siman') ref = h.book + '.' + h.siman + '.' + h.seif;
  if (h.start === 'link' && h.link) ref = decodeURIComponent(h.link.replace(/^https:\/\/[^/]+\//, '').split(/[?#]/)[0]).replace(/_/g, ' ');
  const props = PropertiesService.getScriptProperties(), pk = 'hal_' + id;
  let st = parseJson_(props.getProperty(pk));
  let ver = null;
  if (!st || st.sig !== sig || st.day > day) {   // הגדרה חדשה – מתחילים מנקודת ההתחלה
    const first = halSection_(ref, '');
    ver = halVersion_(first.index || h.book);
    if (!ver.vt) throw new Error('לא נמצאה בספריא מהדורה עברית של הספר עם רישיון לשימוש מסחרי' + (ver.none ? ': ' + ver.none : ''));
    const sec = halSection_(first.ref, ver.vt);
    const isSeg = first.depth && first.sections.length >= first.depth;   // הקישור / ההתחלה מצביעים על סעיף מסוים
    st = { sig, vt: ver.vt, lic: ver.lic, index: first.index || h.book, first: sec.ref, ref: sec.ref, idx: isSeg ? Math.max(0, first.sections[first.sections.length - 1] - 1) : 0, day: start };
  }
  // התקדמות עד היום
  let left = h.per * daysBetween_(st.day, day), guard = 0;
  while (left > 0 && guard++ < 3000) {
    const sec = halSection_(st.ref, st.vt), remain = sec.segs.length - st.idx;
    if (left < remain) { st.idx += left; left = 0; break; }
    left -= Math.max(0, remain); st.ref = sec.next || st.first; st.idx = 0;
  }
  st.day = day;
  props.setProperty(pk, JSON.stringify(st));
  // ההלכות של היום
  const items = []; let r = st.ref, i = st.idx; guard = 0;
  while (items.length < h.per && guard++ < 50) {
    const sec = halSection_(r, st.vt);
    if (i < sec.segs.length) { if (sec.segs[i]) items.push({ label: (sec.heRef || '') + (sec.segs.length > 1 ? ':' + heb_(i + 1) : ''), text: sec.segs[i].slice(0, 4000) }); i++; }
    else { r = sec.next || st.first; i = 0; }
  }
  return { ok: true, title: HAL_BOOKS[h.book] || st.index, items, source: 'ספריא · ' + st.vt + (st.lic ? ' (' + st.lic + ')' : '') };
}
function halHebcal_(h, day) {
  const r = JSON.parse(UrlFetchApp.fetch('https://www.hebcal.com/hebcal?v=1&cfg=json&start=' + day + '&end=' + day + '&' + h.key + '=on&lg=s', { muteHttpExceptions: true }).getContentText());
  const it = (r.items || [])[0];
  if (!it) throw new Error('לא נמצא לימוד להיום ב-Hebcal');
  let ref = '';
  if (/sefaria\.org/.test(it.link || '')) ref = decodeURIComponent(String(it.link).replace(/^https?:\/\/[^/]+\//, '').split(/[?#]/)[0]).replace(/_/g, ' ');
  if (!ref) throw new Error('ל-Hebcal אין קישור לספריא עבור ' + (it.title || h.key));
  const first = halSection_(ref, '');
  const ver = halVersion_(first.index || ref.split(/[.,]/)[0]);
  if (!ver.vt) throw new Error('לא נמצאה מהדורה עברית עם רישיון לשימוש מסחרי');
  const rr = sefGet_('texts/' + sefRef_(ref) + '?context=0&commentary=0&pad=0&vhe=' + encodeURIComponent(ver.vt));
  const segs = flat_(rr.he).map(stripHtml_).filter(Boolean);
  return { ok: true, title: HAL_HEBCAL[h.key], items: segs.map((t, i) => ({ label: i === 0 ? (rr.heRef || '') : '', text: t.slice(0, 4000) })), source: 'ספריא · ' + ver.vt + (ver.lic ? ' (' + ver.lic + ')' : '') };
}

}
